+/** Why a report is not confirmed (`unconfirmed.ts`, shared with the map's view). */
+export type { Unconfirmed };
+
+/** A report's search words (folded, as `searchTokens` reads them, each once) and its mark. */
+export interface ReportSearch {
+  words: string[];
+  unconfirmed?: Unconfirmed;
+}
+
+/** What the search needs beyond the stored report: how many lineups a grunt type has today. */
+export interface SearchFacts {
+  /** The lineups of a grunt type's choice (`grunt:water`): two for a type reported apart. */
+  lineupsOf(typeKey: string): number;
+}
+
+/** The words of a Pokémon: its name, its catalogue key's (`rattata_alola`: its form). */
+function pokemonWords(name: string, key: string | null): string[] {
+  return [name, ...(key === null ? [] : [key])];
+}
+
+/** The words of a reward: a Pokémon with its form and costume, an item, Mega Energy. */
+function rewardWords(reward: TaskReward): string[] {
+  if (reward.kind !== 'encounter') return [reward.name];
+  const costumed = reward.costumed === true || wornBy(reward.name) !== undefined;
+  return [...pokemonWords(reward.name, reward.key), ...(costumed ? ['costumed', 'costume'] : [])];
+}
+
+/** The words of research: its task's text and every reward it stands for. */
+function researchWords(report: ResearchReport): string[] {
+  return [report.research.text, ...report.research.rewards.flatMap(rewardWords)];
+}
+
+/**
+ * The words of Team GO Rocket: who by name (`Water-type grunt`, `Arlo`, `Giovanni`, `Decoy
+ * grunt`), a lineup's first slot, the Pokémon that can be caught, `rocket`, and a leader's
+ * `leader`.
+ */
+function rocketWords(report: RocketReport): string[] {
+  const who = parseRocketKey(report.subjectKey);
+  return [
+    report.rocket.name,
+    report.rocket.lineup ?? '',
+    ...report.rocket.catches.flatMap((caught) => pokemonWords(caught.name, caught.key)),
+    'rocket',
+    ...(who?.kind === 'leader' ? ['leader'] : []),
+  ];
+}
+
+/** Why research is not confirmed: "one of" its rewards, or no reward said; else confirmed. */
+function researchMark(report: ResearchReport): Unconfirmed | undefined {
+  const { rewards } = report.research;
+  if (rewards.length === 0) return { kind: 'no-reward' };
+  if (!isOneOf(report)) return undefined;
+  return {
+    kind: 'one-of',
+    count: rewards.length,
+    encounters: rewards.every((reward) => reward.kind === 'encounter'),
+  };
+}
+
+/**
+ * A grunt of a type reported without its lineup ("not sure which", plan 8R-4L), where the type
+ * has two lineups today: only such a type can be unconfirmed.
+ */
+function rocketMark(report: RocketReport, facts: SearchFacts): Unconfirmed | undefined {
+  const who = parseRocketKey(report.subjectKey);
+  if (who?.kind !== 'grunt' || who.lineup !== undefined) return undefined;
+  const count = facts.lineupsOf(report.subjectKey);
+  return count >= 2 ? { kind: 'not-sure-which', count } : undefined;
+}
+
+/**
+ * A report's search words and whether it is not confirmed (plan 8M-12): research by its task's
+ * words and every reward's name (a Pokémon with its form and costume, an item, Mega Energy), Team
+ * GO Rocket by who, the lineup's Pokémon, `rocket` and a leader's `leader`; never its reporter or
+ * its stop. Research "one of" or without its reward, and a grunt of two lineups "not sure which",
+ * are not confirmed.
+ */
+export function reportSearch(report: StopReport, facts: SearchFacts): ReportSearch {
+  const texts = isResearchReport(report) ? researchWords(report) : rocketWords(report);
+  const words = [...new Set(texts.flatMap(searchTokens))];
+  const unconfirmed = isResearchReport(report) ? researchMark(report) : rocketMark(report, facts);
+  return unconfirmed === undefined ? { words } : { words, unconfirmed };
+}

+/**
+ * The requests of Find one address may make in a minute (plan 8M-12): its own limit, above the
+ * pick lists' 30 (a form asks once), so that typing never empties the group of places. The page
+ * asks once a pause after the text changed.
+ */
+export const FIND_PER_MINUTE = 120;
+
+/** What Find asks: the text (two letters at least) and where its member is, or the map's middle. */
+const findQuery = z
+  .object({
+    q: z
+      .string()
+      .max(FIND_MAX_TEXT)
+      .refine((text) => searchTokens(text).join('').length >= FIND_MIN_LETTERS),
+    near: pointSchema.optional(),
+  })
+  .strict();
+
+/**
+ * Whether a hit reads every word typed in its name (`readsEvery`: as typed, as a beginning, or
+ * misspelt within `allowedTypos`; `STOP_WORDS` ask nothing): Find leaves out a guess that misses a
+ * word, as a filter would, and keeps one that only misspells it.
+ */
+function everyWord(text: string, choice: PlaceChoice): boolean {
+  return choice.fit !== 'guess' || readsEvery(searchTokens(text), searchTokens(choice.title));
+}
+
+/** A place found as the page reads it. */
+function foundPlace(choice: PlaceChoice): FoundPlace {
+  return {
+    kind: choice.kind satisfies FoundPlaceKind,
+    id: choice.id,
+    name: choice.title,
+    lat: choice.point.lat,
+    lon: choice.point.lon,
+  };
+}
+
+/**
+ * Find's places on the live map (plan 8M-12; design "Where each group is searched"): the gyms,
+ * PokéStops and Power Spots of the server by name (`GET /s/<guild>/api/find-places?q=<text>
+ * [&near=<lat>,<lon>]`), through the place search Discord's commands use (`PlaceFinder.find`: the
+ * gym search and the `WayspotIndex` of each kind, held in memory, limited to the server's regions;
+ * no database query), in its order, places that fit alike the nearer to `near` first, at most
+ * `FOUND_PLACES_PER_KIND` of each kind; a guess that misses a word typed is left out. Members only
+ * (`memberOrFailure`, as `/api/stops`), `FIND_PER_MINUTE` per address.
+ */
+export function findRoutes(scope: FastifyInstance, deps: WebDeps): void {
+  const { sessions, parties } = deps;
+  const limiter = new RateLimiter({ uses: FIND_PER_MINUTE, perMs: 60_000 });
+  scope.get(
+    '/api/find-places',
+    { onRequest: limitedPerAddress(limiter, { now: deps.now }) },
+    async (request: GuildRequest, reply) => {
+      void reply.header('cache-control', 'private, no-store');
+      const session = await sessions.of(request, reply);
+      if (!session) return sendFailure(request, reply, 401);
+      const query = findQuery.safeParse(request.query ?? {});
+      if (!query.success) return sendFailure(request, reply, 400);
+      const { guild } = request.params;
+      const refused = await memberOrFailure(deps, request, reply, guild, session.userId);
+      if (refused) return refused;
+      const row = await parties.guilds.idFor(guild);
+      const { q, near } = query.data;
+      const found =
+        row === undefined
+          ? []
+          : deps.stops.ctx.placeFinder.find(
+              row,
+              undefined,
+              q,
+              ALL_PLACE_KINDS,
+              near === undefined ? {} : { anchors: [near] },
+            );
+      const counts = new Map<string, number>();
+      const places: FoundPlace[] = [];
+      for (const choice of found) {
+        const count = counts.get(choice.kind) ?? 0;
+        if (count >= FOUND_PLACES_PER_KIND || !everyWord(q, choice)) continue;
+        counts.set(choice.kind, count + 1);
+        places.push(foundPlace(choice));
+      }
+      const view: FoundPlacesView = { places };
+      return view;
+    },
+  );
+}

+ * answer for a text no longer in the box is dropped. Under two letters there is no search: the
+ * list is as it was. Enter searches at once (`now`), and asks for the places again where they
+ * failed (review F4).
+ */

+  private timer: ReturnType<typeof setTimeout> | undefined;
+  private current: FindState | undefined;
+  /** The text in the box. */
+  private text = '';
+  /** Enter was pressed before the places came for this text. */
+  private picking: string | undefined;
+  /** The places' question asked last: an answer to an earlier one is dropped. */
+  private asked = 0;
+
+  constructor(private readonly deps: FindSearchDeps) {}
+
+  /** The search shown; none while the box holds under two letters. */
+  get state(): FindState | undefined {
+    return this.current;
+  }
+
+  /** The box's text changed. */
+  typed(text: string): void {
+    this.text = text;
+    this.picking = undefined;
+    if (this.timer !== undefined) clearTimeout(this.timer);
+    this.timer = undefined;
+    if (typedWords(text) === undefined) {
+      if (this.current !== undefined) {
+        this.current = undefined;
+        this.deps.changed(undefined, 'ended');
+      }
+      return;
+    }
+    if (this.current !== undefined && this.current.text === text) return;
+    this.timer = setTimeout(() => {
+      this.timer = undefined;
+      this.search();
+    }, FIND_PAUSE_MS);
+  }
+
+  /**
+   * Enter: the search at once where the pause still runs, the places asked for again where they
+   * failed (review F4); true where there is a search.
+   */
+  now(): boolean {
+    if (this.timer !== undefined) {
+      clearTimeout(this.timer);
+      this.timer = undefined;
+      this.search();
+    } else if (this.current?.places === 'failed') {
+      this.current = { ...this.current, places: 'waiting' };
+      this.deps.changed(this.current, 'settled');
+      this.ask(this.current.text);
+    }
+    return this.current !== undefined;
+  }
+
+  /** Enter found no line yet: the first place, once the places come for this text. */
+  pickWhenFound(): void {
+    if (this.current?.places === 'waiting') this.picking = this.current.text;
+  }
+
+  private search(): void {
+    const { text } = this;
+    const words = typedWords(text);
+    if (words === undefined) return;
+    this.current = { text, words, places: 'waiting' };
+    this.deps.changed(this.current, 'began');
+    this.ask(text);
+  }
+
+  /** The places of a text asked for, its answer kept while it is the last question's. */
+  private ask(text: string): void {
+    this.asked += 1;
+    const question = this.asked;
+    const settle = (places: PlacesFound): void => {
+      // An answer for a text no longer in the box, or to an earlier question, is dropped.
+      if (this.current?.text !== text || question !== this.asked) return;
+      this.current = { ...this.current, places };
+      this.deps.changed(this.current, 'settled');
+      if (this.picking === text) {
+        this.picking = undefined;
+        this.deps.pick?.(this.current);
+      }
+    };
+    this.deps.places(text).then(settle, () => {
+      settle('failed');
+    });
+  }
